<script>
const K_ID = 'gcp_id';
const K_SECRET = 'gcp_secret';
const K_COUNT = 'gcp_count';

function randStr(){
  return Math.random().toString(36).slice(2,10)
       +Math.random().toString(36).slice(2,10);
}

function init(){
  let id = localStorage.getItem(K_ID);
  let sec = localStorage.getItem(K_SECRET);
  let cnt = parseInt(localStorage.getItem(K_COUNT) || -1);

  // 创始人 GCP-0
  if(cnt === -1){
    id = 'GCP-0';
    sec = randStr();
    cnt = 0;
    localStorage.setItem(K_ID, id);
    localStorage.setItem(K_SECRET, sec);
    localStorage.setItem(K_COUNT, cnt);
  }

  // 新用户
  if(!id){
    cnt++;
    id = 'GCP-'+cnt;
    sec = randStr();
    localStorage.setItem(K_ID, id);
    localStorage.setItem(K_SECRET, sec);
    localStorage.setItem(K_COUNT, cnt);
  }

  // 显示
  document.getElementById('gid').innerText = id;
  document.getElementById('total').innerText = '全球共同建设者：'+(cnt+1)+' 人';

  let keyArea = document.getElementById('keyArea');
  let secretEl = document.getElementById('secret');
  secretEl.innerText = sec;
  keyArea.style.display = 'block';
}

init();
</script>
